\section{Backend (Node.js/Express \& Logic Engine)}

\subsection{Vai trò và Nhiệm vụ}
\quad Trong hệ thống TourXport, thành phần Backend đóng vai trò là "xương sống" kỹ thuật, chịu trách nhiệm vận hành toàn bộ các logic nghiệp vụ phức tạp và đảm bảo sự kết nối thông suốt giữa lớp ứng dụng (Frontend) và các dịch vụ dữ liệu/AI theo kiến trúc microservices \cite{microservices}. Các nhiệm vụ chính bao gồm:

\begin{itemize}
    \item \textbf{Thiết kế và quản lý API:} Xây dựng hệ thống RESTful API \cite{restapi} cung cấp các điểm cuối (endpoints) cho ứng dụng di động truy xuất dữ liệu địa điểm, người dùng và hành trình.
    \item \textbf{Quản lý Cơ sở dữ liệu:} Thực hiện các thao tác CRUD (Create, Read, Update, Delete) trên MongoDB thông qua Mongoose ODM \cite{mongoose}, đảm bảo tính toàn vẹn và tối ưu hóa truy vấn.
    \item \textbf{Xác thực và Bảo mật:} Triển khai cơ chế bảo mật dựa trên JSON Web Token (JWT) để quản lý phiên đăng nhập và bảo vệ các tài nguyên nhạy cảm của người dùng.
    \item \textbf{Tích hợp dịch vụ AI:} Đóng vai trò trung gian tiếp nhận yêu cầu từ người dùng, gửi dữ liệu ngữ cảnh (Context) cho dịch vụ AI và xử lý kết quả trả về trước khi lưu trữ.
\end{itemize}

\subsection{Tech Stack \& Thư viện}
Hệ thống Backend được xây dựng trên các nền tảng công nghệ hiện đại, ưu tiên tính phi tập trung và hiệu năng xử lý bất đồng bộ:

\begin{itemize}
    \item \textbf{Runtime Environment:} Node.js - cho phép xử lý số lượng lớn các kết nối đồng thời với mô hình Non-blocking Input/Output.
    \item \textbf{Web Framework:} Express.js - tối giản, linh hoạt, hỗ trợ mạnh mẽ việc xây dựng các Middleware xử lý lỗi và xác thực.
    \item \textbf{Database ODM:} Mongoose \cite{mongoose} - hỗ trợ định nghĩa Schema chặt chẽ cho MongoDB và quản lý các mối quan hệ dữ liệu phức tạp.
    \item \textbf{Bảo mật:} \texttt{bcryptjs} \cite{bcryptjs} (mã hóa mật khẩu) và \texttt{jsonwebtoken} (xác thực người dùng).
    \item \textbf{Quản lý môi trường:} Docker \& Docker Compose - đóng gói toàn bộ dịch vụ để đảm bảo tính nhất quán giữa các môi trường phát triển và triển khai.
\end{itemize}

\subsection{Quy trình xử lý}
Quy trình xử lý một yêu cầu điển hình tại Backend được thực hiện qua các giai đoạn nghiêm ngặt:

\begin{enumerate}
    \item \textbf{Tiếp nhận và Phân loại (Routing):} Yêu cầu từ Client được định tuyến đến Controller tương ứng dựa trên URL và phương thức HTTP (GET, POST, PUT, DELETE, PATCH).
    \item \textbf{Kiểm soát truy cập (Middleware):} Các yêu cầu được đưa qua lớp kiểm tra Token. Nếu hợp lệ, thông tin người dùng được trích xuất từ JWT để sử dụng cho các bước tiếp theo.
    \item \textbf{Thực thi Logic nghiệp vụ:} Controller thực hiện các tính toán hoặc gọi đến các Model để truy xuất dữ liệu. Trong trường hợp tạo tour, Backend sẽ giao tiếp với AI Backend để nhận kết quả lịch trình.
    \item \textbf{Quản lý lỗi tập trung (Global Error Handling):} Mọi lỗi phát sinh trong quá trình xử lý đều được bắt bởi một Middleware chuyên trách, đảm bảo server không bị crash và trả về phản hồi chuẩn JSON cho người dùng.
    \item \textbf{Phản hồi kết quả:} Kết quả sau khi xử lý được trả về dưới dạng JSON với cấu trúc chuẩn bao gồm trạng thái (\textit{success}), thông điệp (\textit{message}) và dữ liệu (\textit{data}).
\end{enumerate}

\section{AI Worker (LLM \& RAG Engine)}

\subsection{Vai trò và Nhiệm vụ}

\quad AI Worker là thành phần xử lý trí tuệ nhân tạo cốt lõi của hệ thống TourXport, chịu trách nhiệm diễn giải các yêu cầu du lịch phức tạp của người dùng (thông qua văn bản tự nhiên, ngân sách, thời gian) và tự động lập kế hoạch lịch trình chi tiết (Itinerary). Việc tách biệt AI Worker thành một dịch vụ độc lập (Microservice) khỏi Backend chính giúp tối ưu hóa việc xử lý các tác vụ tốn kém thời gian và tài nguyên tính toán, đặc biệt là quá trình giao tiếp mạng bất đồng bộ với các mô hình ngôn ngữ lớn (LLMs). 

\quad Thành phần này đóng vai trò như một bộ điều phối (Orchestrator), quyết định việc sinh lịch trình tự do bằng LLM thuần túy (Pure LLM) hoặc kết hợp với công nghệ truy xuất dữ liệu (RAG - Retrieval-Augmented Generation) từ cơ sở dữ liệu vector. Trong tương lai, nếu triển khai các tính năng khác cần xử dụng LLM làm cốt lõi, module AI Worker cũng sẽ đóng vai trò là điều phối.

\subsection{Tech Stack \& Thư viện}

\begin{itemize}
    \item \textbf{Ngôn ngữ nền tảng:} Python 3.10.
    \item \textbf{Framework:} FastAPI (phiên bản 0.136.1) \cite{fastapi} – cung cấp khả năng xử lý bất đồng bộ (async) với hiệu suất cao và tự động hóa sinh tài liệu giao diện API.
    \item \textbf{LLM Integration:} Thư viện openai (phiên bản 2.36.0) cho python \cite{openai} – giao tiếp trực tiếp với mô hình gpt-4o-mini. Hệ thống sử dụng phương thức phân tích cú pháp nâng cao để đảm bảo đầu ra luôn tuân thủ nghiêm ngặt chuẩn cấu trúc JSON.
    \item \textbf{Data Validation \& Serialization:} Thư viện pydantic (phiên bản 2.13.4) \cite{pydantic} – định nghĩa các schema ràng buộc kiểu dữ liệu cho Request và Response. Hệ thống tận dụng Pydantic để tự động ánh xạ cấu trúc dữ liệu snake\_case của Python sang định dạng camelCase chuẩn hóa cho luồng giao tiếp với Frontend.
    \item \textbf{Authentication:} Thư viện PyJWT \cite{jwt} – dùng để giải mã và xác thực token JWT được cấp phát chéo từ Backend chính.
\end{itemize}

\subsection{Quy trình xử lý}
\begin{itemize}
    \item \textbf{Bước 1}: Tiếp nhận và Xác thực Yêu cầu — Lớp Controller tại file src/api/endpoints/trip\_generator.py (hàm generate\_trip) tiếp nhận POST request chứa dữ liệu TripGenerateRequest. Tại đây, hệ thống thực thi hàm dependency verify\_token (được định nghĩa trong src/core/security.py) để tiến hành giải mã chuỗi JWT truyền vào thông qua Header, nhằm xác minh danh tính và phiên thao tác của người dùng.
    \item \textbf{Bước 2}: Điều phối Luồng xử lý (Orchestrator) — Sau khi xác thực, dữ liệu được chuyển cho lớp TripOrchestrator tại src/services/trip\_generator/ai\_orchestrator.py (hàm execute). Trong phạm vi giai đoạn thực thi POC hiện tại, hàm này quyết định chuyển hướng luồng xử lý thẳng đến module sinh kế hoạch bằng LLM thuần túy (generate\_itinerary\_pure\_llm), do hệ thống cơ sở dữ liệu địa điểm (Vector Database) vẫn đang trong quá trình xây dựng và cào dữ liệu. Các logic tương lai dành cho RAG (như Vector Search qua DBService) đã được thiết kế sẵn cấu trúc (placeholder) nhằm chờ tích hợp.
    \item \textbf{Bước 3}: Giao tiếp LLM với Structured Output — Lớp AIService tại src/services/trip\_generator/ai\_service.py tiếp nhận tham số đầu vào (ngân sách, số ngày, sở thích) để cấu trúc thành System Prompt và User Prompt. Hệ thống gọi hàm API của OpenAI thông qua phương thức beta.chat.completions.parse, đồng thời truyền vào tham số response\_format=TripGeneratorResponse. Kỹ thuật này ép buộc mô hình gpt-4o-mini không trả về văn bản tự do, mà phải trả về chính xác cấu trúc dữ liệu định nghĩa tại src/schemas/response/trip\_generator.py (bao gồm các thực thể lồng nhau như TripData, DailyItinerary, và Location).
    \item \textbf{Bước 4}: Chuẩn hóa và Phản hồi — Kết quả JSON trả về từ OpenAI lập tức được Pydantic parse thành đối tượng Python nội bộ. Hệ thống chủ động gán cờ is\_from\_db = False để minh bạch hóa với Frontend rằng dữ liệu này hoàn toàn do AI khởi tạo (ảo hóa), các thuộc tính như place\_id sẽ nhận giá trị null. Cuối cùng, FastAPI tự động serialize đối tượng này và phản hồi về cho phía Client (Flutter).
\end{itemize}

\section{Thu thập Dataset (Data Crawler/Scraper)}

\subsection{Vai trò và Nhiệm vụ}

\subsubsection{Vai trò}

\quad Module Crawler/Scraper đóng vai trò là lớp thu thập dữ liệu đầu tiên của hệ thống. Thay vì cào dữ liệu HTML trực tiếp, nhóm sử dụng RapidAPI để bảo đảm ổn định, hợp pháp và có cấu trúc dữ liệu rõ ràng.

\quad Vai trò chính của module này gồm:
\begin{itemize}
    \item \textbf{Thu thập dữ liệu}: Tự động hóa việc truy xuất thông tin về địa điểm (khách sạn, nhà hàng, điểm du lịch), các hình ảnh minh họa và các bài đánh giá (reviews) của người dùng từ hệ thống TripAdvisor.
    \item \textbf{Tiền xử lý dữ liệu}: Chuẩn hóa và lưu trữ dữ liệu thu thập được dưới định dạng JSON theo định dạng mong muốn, sẵn sàng cho việc phân tích và huấn luyện mô hình AI trong tương lai.
\end{itemize}

\subsubsection{Nhiệm vụ}

\quad Nhiệm vụ chính trong giai đoạn POC gồm:
\begin{itemize}
    \item Kết nối và tích hợp API key của RapidAPI để gọi các endpoint của Travel Advisor như \texttt{locations/search} và \texttt{photos/list}.
    \item Lấy các trường dữ liệu cốt lõi như tên địa điểm, ID địa điểm, điểm đánh giá, nội dung review và ảnh minh họa.
    \item Sử dụng các tham số đầu vào của endpoint để lấy đủ số lượng địa điểm, ảnh và đánh giá thay vì chỉ lấy trang đầu tiên.
    \item Xử lý và tối ưu gọi RapidAPI bằng cách giới hạn lưu lượng requests (x requests/ giây, y requests/ tháng) để tránh lỗi HTTP 429 (Too Many Requests) và tránh lãng phí tokens RapidAPI.
    \item Chuyển đổi dữ liệu thô thành dữ liệu JSON tối ưu và lưu trữ cục bộ.
\end{itemize}

\subsection{Tech Stack \& Thư viện}

\begin{itemize}
    \item \textbf{Ngôn ngữ lập trình}: Python 3.9+.
    \item \textbf{Thư viện chính}: 
    \begin{itemize}
        \item requests: Thực hiện các HTTP GET requests đến endpoint của RapidAPI
        \item json: phân tích dữ liệu trả về từ API.
        \item time: Tạo độ trễ giữa các lần gọi API, tuân thủ Rate Limit của gói Rapid API đang sử dụng.
        \item python-dataenv: Quản lý biến môi trường, bảo mật thông tin X-RapidAPI-Key và X-RapidAPI-Host thay vì hard-code trực tiếp vào mã nguồn.
    \end{itemize}
\end{itemize}

\subsection{Quy trình xử lý}

\begin{itemize}
    \item \textbf{Bước 1: Khởi tạo \& Xác thực}
    \begin{itemize}
        \item Load cấu hình từ file .env (API key, Host URL)
        \item Thiết lập HTTP Headers theo yêu cầu của RapidAPI (bao gồm X-RapidAPI-Key và X-RapidAPI-Host)
    \end{itemize}
    \item \textbf{Bước 2: Tìm kiếm ID địa điểm}. 
    \begin{itemize}
        \item Gọi endpoint locations/search sử dụng biến số từ khóa địa điểm để lấy dữ liệu trường location\_id.
        \item Trích xuất và lưu lại location\_id của địa điểm cần lấy data
    \end{itemize}
    \item \textbf{Bước 3: Thu thập thông tin điểm du lịch}. 
    \begin{itemize}
        \item Sử dụng location\_id làm biến số gọi endpoint attraction/list, thu thập thông tin các điểm du lịch lân cận, truy xuất lấy location\_id của từng điểm
        \item Sử dụng location\_id của từng điểm gọi endpoint attration/get-details để lấy thông tin chi tiết
        \item Lọc qua cấu trúc file JSON dữ lại những trường cốt lõi, bỏ những trường dư thừa không sử dụng.
        \item Cộng dồn offset để quản lý danh sách theo trang.
        \item Trong quá trình hệ thống vận hành, kiểm tra HTTP Status Code (Nếu 200 OK -> đi tiếp; Nếu 429 -> Tạm dừng hệ thống 10-30 giây và retry; Nếu lỗi khác -> Ghi log và bỏ qua)
        \item Kích hoạt Rate Limiting Rule với time.sleep trước khi gọi trang tiếp theo.
    \end{itemize}
    \item \textbf{Bước 4: Chuẩn hóa \& Lưu trữ}. 
    \begin{itemize}
        \item Ép kiểu các trường thời gian mở và đóng cửa thành dạng giờ trong ngày cũng như nhiều trường khác
        \item Lưu Data thành file .json để bắt đầu phân tích và huấn luyện mô hình
    \end{itemize}
\end{itemize}
